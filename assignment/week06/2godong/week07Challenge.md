# 사전 지식

# [1단계: 서버 접속]

ssh bandit0@bandit.labs.overthewire.org -p 2220 exit

# [2단계: 파일 탐색]

pwd
ls
ls -a
ls -l
cd [디렉토리명]
cd ..

# [3단계: 파일 읽기]

cat [파일명]
cat ./-
cat "spaces in this filename"
file [파일명]

# [4단계: 데이터 필터링]

grep "문자열" [파일명]
sort [파일명]
sort [파일명] | uniq
sort [파일명] | uniq -u
